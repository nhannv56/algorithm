────────────────────────────────────────────────────────────────────────────────
1. BFS (Breadth-First Search)
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ:
  bfs(start):
    dist[start] = 0, dist[other] = INF
    queue.push(start)
    while queue không rỗng:
      u = queue.front(); queue.pop()
      for v in adj[u]:
        if dist[v] == INF:
          dist[v] = dist[u] + 1
          queue.push(v)

C++ CÀI ĐẶT:
  vector<int> adj[N];
  int dist[N];
  void bfs(int s) {
      fill(dist, dist+N, -1);
      queue<int> q;
      dist[s] = 0; q.push(s);
      while (!q.empty()) {
          int u = q.front(); q.pop();
          for (int v : adj[u])
              if (dist[v] == -1) { dist[v] = dist[u]+1; q.push(v); }
      }
  }

LƯU Ý:
  - Multi-source BFS: push tất cả nguồn vào queue lúc đầu với dist=0
  - BFS trên lưới: dx[]={0,0,1,-1}, dy[]={1,-1,0,0}
  - 0-1 BFS (cạnh trọng số 0 hoặc 1): dùng deque, push_front nếu w=0

────────────────────────────────────────────────────────────────────────────────
2. DFS (Depth-First Search)
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ:
  dfs(u):
    visited[u] = true
    for v in adj[u]:
      if not visited[v]:
        dfs(v)

C++ CÀI ĐẶT:
  bool visited[N];
  void dfs(int u) {
      visited[u] = true;
      for (int v : adj[u])
          if (!visited[v]) dfs(v);
  }

  // DFS tính subtree size
  int sz[N];
  void dfs(int u, int par) {
      sz[u] = 1;
      for (int v : adj[u])
          if (v != par) { dfs(v, u); sz[u] += sz[v]; }
  }

LƯU Ý:
  - Phát hiện chu trình: dùng color[] (0=white, 1=gray, 2=black)
  - Tính in-time / out-time cho Euler Tour
  - Stack overflow với N lớn → dùng DFS iterative với explicit stack

3. Dijkstra
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ:
  dijkstra(src):
    dist[src] = 0, dist[other] = INF
    pq.push({0, src})
    while pq không rỗng:
      {d, u} = pq.top(); pq.pop()
      if d > dist[u]: continue   // đã xử lý rồi
      for {v, w} in adj[u]:
        if dist[u] + w < dist[v]:
          dist[v] = dist[u] + w
          pq.push({dist[v], v})

C++ CÀI ĐẶT:
  vector<pair<int,int>> adj[N];  // {neighbor, weight}
  long long dist[N];
  void dijkstra(int s) {
      fill(dist, dist+N, LLONG_MAX);
      priority_queue<pair<long long,int>, vector<pair<long long,int>>,
                     greater<>> pq;
      dist[s] = 0; pq.push({0, s});
      while (!pq.empty()) {
          auto [d, u] = pq.top(); pq.pop();
          if (d > dist[u]) continue;
          for (auto [v, w] : adj[u])
              if (dist[u]+w < dist[v]) {
                  dist[v] = dist[u]+w;
                  pq.push({dist[v], v});
              }
      }
  }

LƯU Ý:
  - Không dùng khi có cạnh âm → Bellman-Ford
  - Để truy vết đường đi: lưu thêm prev[v] = u
  - Dijkstra nhiều nguồn: set dist[src]=0 cho mọi src trước khi push

4. Floyd-Warshall
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ:
  floyd():
    dist[i][j] = trọng số cạnh i→j (INF nếu không có)
    dist[i][i] = 0
    for k = 1..N:
      for i = 1..N:
        for j = 1..N:
          dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j])

C++ CÀI ĐẶT:
  long long dist[N][N];
  void floyd(int n) {
      for (int k = 0; k < n; k++)
          for (int i = 0; i < n; i++)
              for (int j = 0; j < n; j++)
                  if (dist[i][k] != INF && dist[k][j] != INF)
                      dist[i][j] = min(dist[i][j], dist[i][k]+dist[k][j]);
  }

LƯU Ý:
  - Sau khi chạy, dist[i][i] < 0 → tồn tại chu trình âm
  - Khởi tạo: dist[i][j] = INF/2 để tránh overflow khi cộng

5. Bellman-Ford
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ:
  bellman_ford(src):
    dist[src] = 0, dist[other] = INF
    repeat N-1 lần:
      for mỗi cạnh (u, v, w):
        dist[v] = min(dist[v], dist[u] + w)
    // kiểm tra chu trình âm:
    for mỗi cạnh (u, v, w):
      if dist[u] + w < dist[v]: tồn tại chu trình âm

C++ CÀI ĐẶT:
  struct Edge { int u, v, w; };
  vector<Edge> edges;
  long long dist[N];
  bool bellman_ford(int s, int n) {
      fill(dist, dist+n, LLONG_MAX);
      dist[s] = 0;
      for (int i = 0; i < n-1; i++)
          for (auto& [u,v,w] : edges)
              if (dist[u] != LLONG_MAX && dist[u]+w < dist[v])
                  dist[v] = dist[u]+w;
      for (auto& [u,v,w] : edges)
          if (dist[u] != LLONG_MAX && dist[u]+w < dist[v]) return false; // cycle
      return true;
  }

6. Topological Sort (Kahn's Algorithm)
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ:
  topo_sort():
    tính indegree[] cho mỗi đỉnh
    queue ← tất cả đỉnh có indegree = 0
    while queue không rỗng:
      u = queue.front(); queue.pop()
      result.push(u)
      for v in adj[u]:
        indegree[v]--
        if indegree[v] == 0: queue.push(v)
    if len(result) < N: có chu trình

C++ CÀI ĐẶT:
  vector<int> adj[N];
  int indegree[N];
  vector<int> topo_sort(int n) {
      queue<int> q;
      for (int i = 0; i < n; i++)
          if (!indegree[i]) q.push(i);
      vector<int> res;
      while (!q.empty()) {
          int u = q.front(); q.pop();
          res.push_back(u);
          for (int v : adj[u])
              if (--indegree[v] == 0) q.push(v);
      }
      return res; // size < n → có chu trình
  }

7. Minimum Spanning Tree - Kruskal
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ:
  kruskal():
    sort edges theo trọng số tăng dần
    for mỗi cạnh (u, v, w) theo thứ tự:
      if find(u) != find(v):
        union(u, v)
        mst_cost += w
        mst_edges.push({u, v, w})

C++ CÀI ĐẶT:
  struct Edge { int u, v, w; };
  // Dùng kết hợp với DSU (xem mục Union-Find)
  long long kruskal(int n, vector<Edge>& edges) {
      sort(edges.begin(), edges.end(), [](auto& a, auto& b){ return a.w < b.w; });
      long long cost = 0;
      for (auto& [u,v,w] : edges)
          if (find(u) != find(v)) { unite(u,v); cost += w; }
      return cost;
  }

8. Bipartite Graph Check
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ:
  is_bipartite():
    color[] = -1
    for mỗi đỉnh chưa tô màu:
      bfs/dfs, tô màu xen kẽ 0 và 1
      if gặp cạnh nối 2 đỉnh cùng màu: return false
    return true

C++ CÀI ĐẶT:
  int color[N];
  bool check(int s) {
      queue<int> q; q.push(s); color[s] = 0;
      while (!q.empty()) {
          int u = q.front(); q.pop();
          for (int v : adj[u]) {
              if (color[v] == -1) { color[v] = 1-color[u]; q.push(v); }
              else if (color[v] == color[u]) return false;
          }
      }
      return true;
  }

9. Binary Lifting / LCA
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ:
  preprocess():
    dfs để tính depth[] và up[v][0] = cha trực tiếp của v
    for k = 1..LOG:
      for v = 0..N:
        up[v][k] = up[up[v][k-1]][k-1]

  lca(u, v):
    if depth[u] < depth[v]: swap(u, v)
    // nâng u lên cùng độ sâu với v
    for k = LOG-1..0:
      if depth[u] - depth[v] >= (1<<k): u = up[u][k]
    if u == v: return u
    // nhảy đồng thời cho đến khi cha bằng nhau
    for k = LOG-1..0:
      if up[u][k] != up[v][k]: u=up[u][k]; v=up[v][k]
    return up[u][0]

C++ CÀI ĐẶT:
  const int LOG = 20;
  int up[N][LOG], depth[N];
  void dfs(int u, int p, int d) {
      depth[u] = d; up[u][0] = p;
      for (int k = 1; k < LOG; k++) up[u][k] = up[up[u][k-1]][k-1];
      for (int v : adj[u]) if (v != p) dfs(v, u, d+1);
  }
  int lca(int u, int v) {
      if (depth[u] < depth[v]) swap(u,v);
      int diff = depth[u]-depth[v];
      for (int k = 0; k < LOG; k++) if ((diff>>k)&1) u = up[u][k];
      if (u == v) return u;
      for (int k = LOG-1; k >= 0; k--)
          if (up[u][k] != up[v][k]) { u=up[u][k]; v=up[v][k]; }
      return up[u][0];
  }

10. Tree DP
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ (ví dụ: maximum independent set trên cây):
  dp[v][0] = không chọn v
  dp[v][1] = chọn v

  dfs(v, par):
    dp[v][0] = 0
    dp[v][1] = val[v]
    for c in children(v):
      dfs(c, v)
      dp[v][0] += max(dp[c][0], dp[c][1])
      dp[v][1] += dp[c][0]

C++ CÀI ĐẶT:
  int dp[N][2], val[N];
  void dfs(int u, int p) {
      dp[u][0] = 0; dp[u][1] = val[u];
      for (int v : adj[u]) {
          if (v == p) continue;
          dfs(v, u);
          dp[u][0] += max(dp[v][0], dp[v][1]);
          dp[u][1] += dp[v][0];
      }
  }

11. Euler Tour / DFS Order on Tree
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ:
  timer = 0
  dfs(v, par):
    in[v] = ++timer
    for c in children(v): dfs(c, v)
    out[v] = timer
  // subtree của v = đoạn [in[v], out[v]] trên mảng Euler

C++ CÀI ĐẶT:
  int in_time[N], out_time[N], order[N], timer = 0;
  void dfs(int u, int p) {
      in_time[u] = ++timer; order[timer] = u;
      for (int v : adj[u]) if (v != p) dfs(v, u);
      out_time[u] = timer;
  }
  // truy vấn subtree u: range [in_time[u], out_time[u]] trên mảng order[]

12. Segment Tree
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ:
  build(node, l, r):
    if l == r: tree[node] = a[l]; return
    mid = (l+r)/2
    build(2*node, l, mid); build(2*node+1, mid+1, r)
    tree[node] = tree[2*node] + tree[2*node+1]

  query(node, l, r, ql, qr):
    if qr < l or r < ql: return 0
    if ql <= l and r <= qr: return tree[node]
    mid = (l+r)/2
    return query(2*node,l,mid,ql,qr) + query(2*node+1,mid+1,r,ql,qr)

  update(node, l, r, pos, val):
    if l == r: tree[node] = val; return
    mid = (l+r)/2
    if pos <= mid: update(2*node, l, mid, pos, val)
    else: update(2*node+1, mid+1, r, pos, val)
    tree[node] = tree[2*node] + tree[2*node+1]

C++ CÀI ĐẶT:
  long long tree[4*N];
  void build(int node, int l, int r, int* a) {
      if (l == r) { tree[node] = a[l]; return; }
      int mid = (l+r)/2;
      build(2*node,l,mid,a); build(2*node+1,mid+1,r,a);
      tree[node] = tree[2*node] + tree[2*node+1];
  }
  long long query(int node, int l, int r, int ql, int qr) {
      if (qr < l || r < ql) return 0;
      if (ql <= l && r <= qr) return tree[node];
      int mid = (l+r)/2;
      return query(2*node,l,mid,ql,qr) + query(2*node+1,mid+1,r,ql,qr);
  }
  void update(int node, int l, int r, int pos, long long val) {
      if (l == r) { tree[node] = val; return; }
      int mid = (l+r)/2;
      if (pos <= mid) update(2*node,l,mid,pos,val);
      else update(2*node+1,mid+1,r,pos,val);
      tree[node] = tree[2*node] + tree[2*node+1];
  }

LƯU Ý (Lazy Propagation cho range update):
  lazy[node] lưu giá trị cần cập nhật trễ
  khi query/update, push lazy xuống con trước khi đi sâu

13. Fenwick Tree / Binary Indexed Tree (BIT)
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ:
  update(i, delta):
    while i <= N: tree[i] += delta; i += i & (-i)

  query(i):  // prefix sum [1..i]
    sum = 0
    while i > 0: sum += tree[i]; i -= i & (-i)
    return sum

  range_sum(l, r) = query(r) - query(l-1)

C++ CÀI ĐẶT:
  long long tree[N+1];
  void update(int i, long long delta) {
      for (; i <= N; i += i&(-i)) tree[i] += delta;
  }
  long long query(int i) {
      long long s = 0;
      for (; i > 0; i -= i&(-i)) s += tree[i];
      return s;
  }

LƯU Ý:
  - BIT 2D: update(i,j,delta) dùng 2 vòng lặp lồng nhau
  - Đếm inversion: duyệt từ phải sang trái, query(a[i]-1) + update(a[i],1)

14. Sparse Table / RMQ
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ:
  build():
    sparse[0][i] = a[i]
    for k = 1..LOG:
      for i = 0..N-(1<<k):
        sparse[k][i] = min(sparse[k-1][i], sparse[k-1][i+(1<<(k-1))])

  query(l, r):
    k = floor(log2(r - l + 1))
    return min(sparse[k][l], sparse[k][r-(1<<k)+1])

C++ CÀI ĐẶT:
  int sparse[LOG][N], lg[N+1];
  void build(int* a, int n) {
      for (int i = 0; i < n; i++) sparse[0][i] = a[i];
      for (int k = 1; (1<<k) <= n; k++)
          for (int i = 0; i+(1<<k)-1 < n; i++)
              sparse[k][i] = min(sparse[k-1][i], sparse[k-1][i+(1<<(k-1))]);
      lg[1] = 0;
      for (int i = 2; i <= n; i++) lg[i] = lg[i/2]+1;
  }
  int query(int l, int r) {
      int k = lg[r-l+1];
      return min(sparse[k][l], sparse[k][r-(1<<k)+1]);
  }

15. Prefix Sum
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ:
  // 1D
  prefix[0] = 0
  prefix[i] = prefix[i-1] + a[i]
  sum(l, r) = prefix[r] - prefix[l-1]

  // 2D
  prefix[i][j] = a[i][j] + prefix[i-1][j] + prefix[i][j-1] - prefix[i-1][j-1]
  sum(r1,c1,r2,c2) = prefix[r2][c2] - prefix[r1-1][c2]
                   - prefix[r2][c1-1] + prefix[r1-1][c1-1]

C++ CÀI ĐẶT:
  long long pre[N+1];
  void build(int* a, int n) {
      pre[0] = 0;
      for (int i = 1; i <= n; i++) pre[i] = pre[i-1] + a[i];
  }
  long long sum(int l, int r) { return pre[r] - pre[l-1]; }

16. Monotonic Stack / Queue
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ (Next Greater Element - bên phải):
  stack S
  for i = 0..N-1:
    while S không rỗng và a[S.top()] < a[i]:
      nge[S.top()] = i   // phần tử bị pop: a[i] là NGE của nó
      S.pop()
    S.push(i)
  while S không rỗng: nge[S.top()] = N; S.pop()  // không có NGE

GIẢ MÃ (Sliding Window Maximum, window size k):
  deque D  // lưu index, đơn điệu giảm
  for i = 0..N-1:
    while D không rỗng và D.front() < i-k+1: D.pop_front()
    while D không rỗng và a[D.back()] <= a[i]: D.pop_back()
    D.push_back(i)
    if i >= k-1: max_window[i-k+1] = a[D.front()]

C++ CÀI ĐẶT:
  // NGE
  vector<int> nge(N, N);
  stack<int> st;
  for (int i = 0; i < N; i++) {
      while (!st.empty() && a[st.top()] < a[i]) { nge[st.top()] = i; st.pop(); }
      st.push(i);
  }

17. Trie Tree
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ:
  struct Node:
    children[26]
    is_end = false

  insert(s):
    cur = root
    for c in s: nếu chưa có node → tạo mới; cur = cur.children[c-'a']
    cur.is_end = true

  search(s):
    cur = root
    for c in s: nếu không có node → return false; cur = cur.children[c-'a']
    return cur.is_end

C++ CÀI ĐẶT:
  struct Trie {
      int ch[MAXN][26], cnt;
      bool end[MAXN];
      Trie() : cnt(0) { memset(ch[0], -1, sizeof(ch[0])); end[0] = false; }
      void insert(const string& s) {
          int cur = 0;
          for (char c : s) {
              int x = c-'a';
              if (ch[cur][x] == -1) {
                  ch[++cnt][0] = -1; memset(ch[cnt],-1,sizeof(ch[cnt]));
                  end[cnt] = false; ch[cur][x] = cnt;
              }
              cur = ch[cur][x];
          }
          end[cur] = true;
      }
  };

18. Union-Find (DSU)
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ:
  parent[i] = i, rank[i] = 0

  find(x):
    if parent[x] != x: parent[x] = find(parent[x])  // path compression
    return parent[x]

  union(x, y):
    px = find(x), py = find(y)
    if px == py: return false
    if rank[px] < rank[py]: swap(px, py)
    parent[py] = px
    if rank[px] == rank[py]: rank[px]++
    return true

C++ CÀI ĐẶT:
  int par[N], rnk[N];
  void init(int n) { iota(par, par+n, 0); fill(rnk, rnk+n, 0); }
  int find(int x) { return par[x] == x ? x : par[x] = find(par[x]); }
  bool unite(int x, int y) {
      x = find(x); y = find(y);
      if (x == y) return false;
      if (rnk[x] < rnk[y]) swap(x,y);
      par[y] = x;
      if (rnk[x] == rnk[y]) rnk[x]++;
      return true;
  }

20. Dynamic Programming (DP)
────────────────────────────────────────────────────────────────────────────────
PATTERN CHUNG:
  1. Định nghĩa dp[i] hoặc dp[i][j]
  2. Base case
  3. Transition (recurrence)
  4. Thứ tự tính (bottom-up) hoặc memoization (top-down)

VÍ DỤ - Đếm đường đi trên lưới:
  dp[i][j] = số đường đi từ (0,0) đến (i,j)
  dp[0][0] = 1
  dp[i][j] = dp[i-1][j] + dp[i][j-1]  (nếu ô đi được)

VÍ DỤ - DP trên dãy (coin change):
  dp[0] = 0, dp[i] = INF
  for mỗi coin c:
    for i = c..W: dp[i] = min(dp[i], dp[i-c] + 1)

C++ MEMOIZATION:
  map<tuple<int,int>, long long> memo;
  long long solve(int i, int j) {
      if (base case) return ...;
      if (memo.count({i,j})) return memo[{i,j}];
      return memo[{i,j}] = ...;
  }

21. Bitmask DP / TSP
────────────────────────────────────────────────────────────────────────────────
GIẢ MÃ (TSP):
  dp[mask][i] = chi phí nhỏ nhất khi đã thăm tập mask, đang ở i
  dp[1<<start][start] = 0, còn lại = INF

  for mask = 0..(1<<N)-1:
    for i = 0..N-1:
      if (mask >> i) & 1 == 0: continue
      for j = 0..N-1:
        if (mask >> j) & 1: continue
        new_mask = mask | (1<<j)
        dp[new_mask][j] = min(dp[new_mask][j], dp[mask][i] + cost[i][j])

  answer = min over i of (dp[(1<<N)-1][i] + cost[i][start])

  // Fast exponentiation
  pow_mod(base, exp, mod):
    result = 1
    base %= mod
    while exp > 0:
      if exp & 1: result = result * base % mod
      base = base * base % mod
      exp >>= 1
    return result

  inv_fact[N] = pow_mod(fact[N], p-2, p)
  for i=N-1..0: inv_fact[i] = inv_fact[i+1]*(i+1) % p

      while (i<=mid) tmp.push_back(a[i++]);
      while (j<=r) tmp.push_back(a[j++]);
      for (int k=l;k<=r;k++) a[k]=tmp[k-l];
      return cnt;
  }

GIẢ MÃ (hash-based):
  // Tìm centroid để root cây tại centroid
  find_centroid(v, par, n):
    for c in children(v):
      if sz[c] > n/2: return find_centroid(c, v, n)
    return v

  // Hash subtree từ lá lên gốc
  hash_tree(v, par):
    child_hashes = sorted([hash_tree(c, v) for c in children(v)])
    hash[v] = map(child_hashes) to unique integer
    return hash[v]

  // Hai cây isomorphic ↔ hash của centroid bằng nhau

C++ CÀI ĐẶT:
  map<vector<int>, int> canonical;
  int get_hash(int u, int p) {
      vector<int> child_hashes;
      for (int v : adj[u]) if (v != p) child_hashes.push_back(get_hash(v, u));
      sort(child_hashes.begin(), child_hashes.end());
      if (!canonical.count(child_hashes)) canonical[child_hashes] = canonical.size();
      return canonical[child_hashes];
  }
